```python
def sort_bboxes(bboxes, page_width):
    # 将页面分为多个水平区域
    bboxes.sort(key=lambda bbox: bbox[1])  # 按 y_min 排序
    regions = []
    current_region = []
    current_y = bboxes[0][1] if bboxes else 0
    threshold = 50  # 定义一个阈值，用于判断是否属于同一水平区域

    for bbox in bboxes:
        # 超出后就是一个新的区域
        if bbox[1] - current_y > threshold:
            regions.append(current_region)
            current_region = []
            current_y = bbox[1]
        current_region.append(bbox)
    if current_region:
        regions.append(current_region)

    sorted_bboxes = []

    # 在每个水平区域内识别单列和双列部分并排序
    for region in regions:
        single_column_bboxes = []
        double_column_bboxes = []

        column_boundary = page_width / 2

        for bbox in region:
            x_min, y_min, x_max, y_max = bbox
            if x_max <= column_boundary or x_min >= column_boundary:
                double_column_bboxes.append(bbox)
            else:
                single_column_bboxes.append(bbox)

        # 在单列部分按 y_min 排序
        single_column_bboxes.sort(key=lambda bbox: bbox[1])

        # 在双列部分按 y_min 排序，然后按 x_min 排序
        double_column_bboxes.sort(key=lambda bbox: (bbox[1], bbox[0]))

        # 合并排序结果
        sorted_bboxes.extend(single_column_bboxes)
        sorted_bboxes.extend(double_column_bboxes)

    return sorted_bboxes

# 示例 bbox 列表
bboxes = [
    (100, 200, 150, 250),
    (50, 100, 100, 150),
    (200, 200, 250, 250),
    (50, 200, 100, 250),
    (100, 100, 150, 150),
    (300, 100, 350, 150),  # 假设这是双列区域的 bbox
    (300, 200, 350, 250)   # 假设这是双列区域的 bbox
]

# 假设页面宽度为 400
page_width = 400

# 排序 bbox
sorted_bboxes = sort_bboxes(bboxes, page_width)

# 打印排序后的 bbox
for bbox in sorted_bboxes:
    print(bbox)
```

解释

1. 将页面分为多个水平区域：根据 y_min 坐标将页面分为多个水平区域，每个区域内的 bbox 的 y_min 坐标差异不超过阈值 threshold。
2. 识别单列和双列部分：在每个水平区域内，根据 x_min 和 x_max 坐标将 bbox 分为单列和双列部分。
3. 排序：在单列部分按 y_min 排序，在双列部分先按 y_min 排序，再按 x_min 排序。
4. 合并排序结果：将各个部分的排序结果合并，得到最终的阅读顺序。
